cmake_minimum_required(VERSION 3.20)
project(kog_snsf_ares LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
set(ARES ${CMAKE_CURRENT_SOURCE_DIR}/../ares-snsf)
set(PROCESSORS wdc65816 spc700 arm7tdmi gsu hg51b upd96050)
set(SOURCES ares.cpp ${ARES}/nall/nall/instruction-set.cpp nall-path.cpp ${ARES}/ares/sfc/sfc.cpp ${ARES}/libco/libco.c)
if(WIN32)
  list(APPEND SOURCES nall-windows.cpp)
endif()
foreach(CPU IN LISTS PROCESSORS)
  list(APPEND SOURCES ${ARES}/ares/component/processor/${CPU}/${CPU}.cpp)
endforeach()
add_library(kog_ares_snsf STATIC ${SOURCES})
target_include_directories(kog_ares_snsf PUBLIC ${ARES} ${ARES}/ares ${ARES}/nall)
target_compile_definitions(kog_ares_snsf PUBLIC PROFILE_ACCURACY PRIVATE LIBCO_MP)
target_link_libraries(kog_ares_snsf PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(kog_ares_snsf PUBLIC ws2_32 shell32)
endif()
if(NOT MSVC)
 target_compile_options(kog_ares_snsf PRIVATE -fwrapv -Wno-narrowing)
endif()
set(PSFLIB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../psflib" CACHE PATH "psflib source directory")
add_library(kog_snsf_psflib STATIC ${PSFLIB_SOURCE}/psflib.c)
target_link_libraries(kog_snsf_psflib PRIVATE ZLIB::ZLIB)
add_library(kog_snsf_embedded STATIC loader.cpp renderer.cpp runner.cpp)
target_compile_definitions(kog_snsf_embedded PRIVATE KOG_EMBEDDED)
target_link_libraries(kog_snsf_embedded PUBLIC kog_ares_snsf kog_snsf_psflib ZLIB::ZLIB)
option(KOG_SNSF_BUILD_EXECUTABLE "Build the SNSF protocol executable" OFF)
if(KOG_SNSF_BUILD_EXECUTABLE)
 add_executable(kog-snsf-ares main.cpp)
 target_link_libraries(kog-snsf-ares PRIVATE kog_snsf_embedded)
endif()
install(TARGETS kog_snsf_embedded kog_ares_snsf kog_snsf_psflib ARCHIVE DESTINATION lib)
option(KOG_SNSF_BUILD_TESTS "Build native SNSF concurrency test" OFF)
if(KOG_SNSF_BUILD_TESTS AND UNIX)
 add_executable(kog-snsf-concurrency-test concurrency_test.cpp)
 target_link_libraries(kog-snsf-concurrency-test PRIVATE kog_snsf_embedded)
endif()
if(KOG_SNSF_BUILD_TESTS)
 add_executable(kog-snsf-loader-test loader_test.cpp loader.cpp)
 target_link_libraries(kog-snsf-loader-test PRIVATE kog_snsf_psflib ZLIB::ZLIB)
endif()
